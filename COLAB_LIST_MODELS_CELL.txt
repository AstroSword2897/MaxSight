# Change BASE if your checkpoints live elsewhere. Set COPY_TO to copy all best models into one folder.

from pathlib import Path
import shutil

BASE = Path("/content/drive/MyDrive/MaxSight")
COPY_TO = None  # or e.g. Path("/content/drive/MyDrive/MaxSight/best_models") to copy all best_model.pt

if not BASE.exists():
    print("Base not found:", BASE, "- mount Drive and set BASE to your MaxSight folder.")
else:
    dirs = sorted(d for d in BASE.iterdir() if d.is_dir() and d.name.startswith("checkpoints_"))
    if not dirs:
        print("No checkpoints_* folders under", BASE)
    else:
        if COPY_TO is not None:
            COPY_TO.mkdir(parents=True, exist_ok=True)
        print("Condition models (best_model.pt):\n")
        for d in dirs:
            cond = d.name.replace("checkpoints_", "")
            best = d / "best_model.pt"
            last = d / "last_checkpoint.pt"
            if best.exists():
                print(" ", cond)
                print("   best: ", best.resolve())
                if last.exists():
                    print("   last: ", last.resolve())
                if COPY_TO is not None:
                    dest = COPY_TO / f"best_model_{cond}.pt"
                    shutil.copy2(best, dest)
                    print("   copied to:", dest)
                print()
            else:
                print(" ", cond, ": no best_model.pt\n")
        if COPY_TO is not None:
            print("All best models copied to:", COPY_TO.resolve())
