# Downloads Open Images V6, BDD100K, and ADE20K for inference

import os, sys, subprocess
from pathlib import Path
import requests
import zipfile
import shutil

# Setup
BASE_DIR = Path("/content/drive/MyDrive/MaxSight/datasets") if 'COLAB_GPU' in os.environ or 'COLAB_JUPYTER_IP' in os.environ else Path("/content/datasets")
os.makedirs(BASE_DIR, exist_ok=True)

# Install FiftyOne
subprocess.run([sys.executable, "-m", "pip", "install", "-q", "fiftyone"], check=False)
import fiftyone as fo

print("="*70)
print("Downloading Inference Datasets")
print("="*70)

# 1. Open Images V6
print("\n📥 Open Images V6 (~2 GB, 30-60 min)...")
open_images_dir = BASE_DIR / "open_images_v6"
validation_dir = open_images_dir / "validation"
csv_path = open_images_dir / "validation-annotations-bbox.csv"

if not validation_dir.exists() or len(list(validation_dir.rglob("*.jpg"))) < 1000:
    dataset = fo.zoo.load_zoo_dataset("open-images-v6", split="validation", dataset_dir=str(BASE_DIR.parent), label_types=["detections"], max_samples=None)
    print(f"✓ Downloaded {len(dataset)} images")
    
    # Reorganize
    for path in [BASE_DIR.parent / "open-images-v6-validation", BASE_DIR.parent / "open-images-v6/validation"]:
        if path.exists():
            validation_dir.mkdir(parents=True, exist_ok=True)
            for img in path.rglob("*.jpg"):
                img_id = img.stem
                dest = validation_dir / img_id[:2] / img.name if len(img_id) >= 2 else validation_dir / img.name
                dest.parent.mkdir(exist_ok=True)
                if not dest.exists():
                    img.rename(dest)
            break
    
    # Download CSV
    r = requests.get("https://storage.googleapis.com/openimages/v6/oidv6-validation-annotations-bbox.csv", stream=True)
    with open(csv_path, 'wb') as f:
        for chunk in r.iter_content(8192):
            f.write(chunk)
    print("✓ Annotations downloaded")

oi6_count = len(list(validation_dir.rglob("*.jpg"))) if validation_dir.exists() else 0
print(f"✅ Open Images V6: {oi6_count:,} images")

# 2. BDD100K
print("\n📥 BDD100K (~600 MB)...")
bdd_dir = BASE_DIR / "bdd100k"
images_dir = bdd_dir / "images/100k/val"
labels_dir = bdd_dir / "labels"
labels_path = labels_dir / "bdd100k_labels_images_val.json"

if not images_dir.exists() or not labels_path.exists():
    try:
        # Labels
        r = requests.get("https://dl.cv.ethz.ch/bdd100k/data/bdd100k_det_20_labels_trainval.zip", stream=True)
        labels_zip = bdd_dir / "labels.zip"
        labels_dir.mkdir(parents=True, exist_ok=True)
        with open(labels_zip, 'wb') as f:
            for chunk in r.iter_content(8192):
                f.write(chunk)
        
        with zipfile.ZipFile(labels_zip, 'r') as z:
            z.extractall(bdd_dir / "temp")
        for p in (bdd_dir / "temp").rglob("det_val.json"):
            shutil.copy(p, labels_path)
            break
        shutil.rmtree(bdd_dir / "temp", ignore_errors=True)
        labels_zip.unlink()
        
        # Images
        r = requests.get("https://dl.cv.ethz.ch/bdd100k/data/100k_images_val.zip", stream=True)
        images_zip = bdd_dir / "images.zip"
        with open(images_zip, 'wb') as f:
            for chunk in r.iter_content(8192):
                f.write(chunk)
        
        with zipfile.ZipFile(images_zip, 'r') as z:
            z.extractall(bdd_dir / "temp")
        for p in (bdd_dir / "temp").rglob("val"):
            if p.is_dir() and len(list(p.glob("*.jpg"))) > 0:
                images_dir.mkdir(parents=True, exist_ok=True)
                if images_dir.exists():
                    shutil.rmtree(images_dir)
                shutil.move(str(p), str(images_dir))
                break
        shutil.rmtree(bdd_dir / "temp", ignore_errors=True)
        images_zip.unlink()
        print("✅ BDD100K downloaded")
    except Exception as e:
        print(f"⚠ BDD100K failed: {e}")

bdd_count = len(list(images_dir.glob("*.jpg"))) if images_dir.exists() else 0
print(f"✅ BDD100K: {bdd_count:,} images")

# 3. ADE20K
print("\n📥 ADE20K (~1 GB)...")
ade_dir = BASE_DIR / "ade20k"
ade_val_images = ade_dir / "images/validation"

if not ade_val_images.exists() or len(list(ade_val_images.glob("*.jpg"))) < 1000:
    try:
        r = requests.get("http://data.csail.mit.edu/places/ADEchallenge/ADEChallengeData2016.zip", stream=True)
        zip_path = ade_dir / "ade20k.zip"
        ade_dir.mkdir(parents=True, exist_ok=True)
        with open(zip_path, 'wb') as f:
            for chunk in r.iter_content(8192):
                f.write(chunk)
        
        with zipfile.ZipFile(zip_path, 'r') as z:
            z.extractall(ade_dir / "temp")
        
        ade_data = ade_dir / "temp/ADEChallengeData2016"
        if (ade_data / "images").exists():
            shutil.move(str(ade_data / "images"), str(ade_dir / "images"))
        if (ade_data / "annotations").exists():
            shutil.move(str(ade_data / "annotations"), str(ade_dir / "annotations"))
        
        shutil.rmtree(ade_dir / "temp", ignore_errors=True)
        zip_path.unlink()
        print("✅ ADE20K downloaded")
    except Exception as e:
        print(f"⚠ ADE20K failed: {e}")

ade_count = len(list(ade_val_images.glob("*.jpg"))) if ade_val_images.exists() else 0
print(f"✅ ADE20K: {ade_count:,} images")

# Summary
print("\n" + "="*70)
print(f"📁 Datasets saved to: {BASE_DIR}")
print(f"✅ Open Images V6: {oi6_count:,} images")
print(f"✅ BDD100K: {bdd_count:,} images")
print(f"✅ ADE20K: {ade_count:,} images")
print("="*70)
