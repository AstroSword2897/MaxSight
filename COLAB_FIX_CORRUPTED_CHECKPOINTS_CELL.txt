# Run this in Colab to detect and fix corrupted checkpoints
# Paste into a Colab cell and run (after mounting Drive)

from pathlib import Path
import torch
import json

BASE = Path("/content/drive/MyDrive/MaxSight")
corrupted = []
fixed = []

print("="*70)
print("Checking for corrupted checkpoints...")
print("="*70)

for d in sorted(BASE.iterdir()):
    if not d.is_dir() or not d.name.startswith("checkpoints_"):
        continue
    
    cond = d.name.replace("checkpoints_", "")
    last_ckpt = d / "last_checkpoint.pt"
    best_ckpt = d / "best_model.pt"
    
    # Check last_checkpoint.pt
    if last_ckpt.exists():
        try:
            ckpt = torch.load(str(last_ckpt), map_location="cpu", weights_only=True)
            if isinstance(ckpt, dict) and "epoch" in ckpt:
                print(f"✅ {cond}: last_checkpoint.pt OK (epoch {ckpt.get('epoch', '?')})")
            else:
                print(f"⚠️  {cond}: last_checkpoint.pt exists but missing 'epoch' key")
        except (EOFError, RuntimeError, Exception) as e:
            print(f"❌ {cond}: last_checkpoint.pt CORRUPTED ({type(e).__name__})")
            corrupted.append((cond, last_ckpt, str(e)))
            
            # Try to fix: if best_model.pt exists, copy it as last_checkpoint.pt
            if best_ckpt.exists():
                try:
                    best = torch.load(str(best_ckpt), map_location="cpu", weights_only=True)
                    # Create a minimal last_checkpoint from best_model
                    if isinstance(best, dict):
                        # If best_model has epoch info, use it; otherwise set to 0
                        epoch = best.get("epoch", 0)
                        fixed_ckpt = {
                            "epoch": epoch,
                            "model_state_dict": best.get("model_state_dict", best),
                            "optimizer_state_dict": best.get("optimizer_state_dict", {}),
                            "best_val_loss": best.get("best_val_loss", float("inf")),
                        }
                    else:
                        fixed_ckpt = {
                            "epoch": 0,
                            "model_state_dict": best,
                            "optimizer_state_dict": {},
                            "best_val_loss": float("inf"),
                        }
                    
                    # Backup corrupted file
                    backup = d / "last_checkpoint.pt.corrupted"
                    last_ckpt.rename(backup)
                    print(f"   → Backed up corrupted file to: {backup.name}")
                    
                    # Save fixed checkpoint
                    torch.save(fixed_ckpt, str(last_ckpt))
                    fixed.append(cond)
                    print(f"   ✅ Fixed: created new last_checkpoint.pt from best_model.pt")
                except Exception as fix_e:
                    print(f"   ⚠️  Could not fix: {fix_e}")
            else:
                print(f"   ⚠️  No best_model.pt to recover from - will train from scratch")
    else:
        print(f"ℹ️  {cond}: no last_checkpoint.pt (will train from scratch)")

print("\n" + "="*70)
if corrupted:
    print(f"Found {len(corrupted)} corrupted checkpoint(s)")
    if fixed:
        print(f"Fixed {len(fixed)} checkpoint(s): {', '.join(fixed)}")
else:
    print("✅ No corrupted checkpoints found!")
print("="*70)
