# Quick verification script - checks that everything is set up correctly
# Run this in Colab to verify your setup before switching to CPU/GPU
# Paste into a Colab cell and run (after mounting Drive)

from pathlib import Path
import torch
import json

print("="*70)
print("QUICK SETUP VERIFICATION")
print("="*70)

# Check 1: Drive mounted
print("\n1️⃣  Checking Google Drive mount...")
drive_path = Path("/content/drive")
if drive_path.exists():
    print("   ✅ Drive is mounted")
else:
    print("   ❌ Drive not mounted!")
    print("   Run: from google.colab import drive; drive.mount('/content/drive')")
    exit(1)

# Check 2: Base checkpoint directory exists
print("\n2️⃣  Checking checkpoint base directory...")
base_ckpt = Path("/content/drive/MyDrive/MaxSight")
if base_ckpt.exists():
    print(f"   ✅ Base directory exists: {base_ckpt}")
else:
    print(f"   ⚠️  Base directory not found: {base_ckpt}")
    print("   This is OK if you haven't trained yet")

# Check 3: Find all checkpoint directories
print("\n3️⃣  Checking for checkpoint directories...")
if base_ckpt.exists():
    checkpoint_dirs = [d for d in base_ckpt.iterdir() 
                      if d.is_dir() and d.name.startswith("checkpoints_")]
    
    if checkpoint_dirs:
        print(f"   ✅ Found {len(checkpoint_dirs)} condition(s):")
        for ckpt_dir in sorted(checkpoint_dirs):
            condition = ckpt_dir.name.replace("checkpoints_", "")
            
            # Check for important files
            best_model = ckpt_dir / "best_model.pt"
            last_ckpt = ckpt_dir / "last_checkpoint.pt"
            history_json = ckpt_dir / "training_history.json"
            resume_info = ckpt_dir / "resume_info.json"
            
            files_found = []
            total_size_mb = 0
            
            if best_model.exists():
                size = best_model.stat().st_size / (1024 * 1024)
                files_found.append(f"best_model.pt ({size:.1f} MB)")
                total_size_mb += size
            
            if last_ckpt.exists():
                size = last_ckpt.stat().st_size / (1024 * 1024)
                files_found.append(f"last_checkpoint.pt ({size:.1f} MB)")
                total_size_mb += size
                # Try to verify it's not corrupted
                try:
                    ckpt = torch.load(str(last_ckpt), map_location="cpu", weights_only=True)
                    epoch = ckpt.get("epoch", "?")
                    print(f"      ✅ {condition}: {', '.join(files_found)} (last_ckpt epoch: {epoch})")
                except (EOFError, RuntimeError) as e:
                    print(f"      ⚠️  {condition}: {', '.join(files_found)} (last_ckpt CORRUPTED: {type(e).__name__})")
            else:
                if files_found:
                    print(f"      ℹ️  {condition}: {', '.join(files_found)} (no last_checkpoint.pt)")
                else:
                    print(f"      ⚠️  {condition}: No checkpoints found")
            
            if history_json.exists():
                try:
                    with open(history_json) as f:
                        history = json.load(f)
                    epochs = len(history.get("epochs", []))
                    print(f"         Training history: {epochs} epochs recorded")
                except:
                    pass
            
            # Count epoch checkpoints
            epoch_ckpts = list(ckpt_dir.glob("checkpoint_epoch_*.pt"))
            if epoch_ckpts:
                print(f"         Epoch snapshots: {len(epoch_ckpts)} files")
    else:
        print("   ℹ️  No checkpoint directories found yet")
        print("   This is OK if you haven't started training")

# Check 4: Data directories
print("\n4️⃣  Checking data directories...")
data_dir = Path("/content/drive/MyDrive/MaxSight_Training")
if data_dir.exists():
    print(f"   ✅ Data directory exists: {data_dir}")
    
    # Check for annotation files
    splits_dir = data_dir / "cleaned_splits"
    if splits_dir.exists():
        train_ann = splits_dir / "maxsight_train.json"
        val_ann = splits_dir / "maxsight_val.json"
        
        if train_ann.exists():
            print(f"   ✅ Train annotations: {train_ann}")
        else:
            print(f"   ⚠️  Train annotations not found: {train_ann}")
        
        if val_ann.exists():
            print(f"   ✅ Val annotations: {val_ann}")
        else:
            print(f"   ⚠️  Val annotations not found: {val_ann}")
    else:
        print(f"   ⚠️  Splits directory not found: {splits_dir}")
else:
    print(f"   ⚠️  Data directory not found: {data_dir}")

# Check 5: Repo setup
print("\n5️⃣  Checking repo setup...")
repo_path = Path("/content/2026-Prototype")
if repo_path.exists():
    print(f"   ✅ Repo exists: {repo_path}")
    
    train_script = repo_path / "scripts" / "train_maxsight.py"
    if train_script.exists():
        print(f"   ✅ Training script found: {train_script}")
    else:
        print(f"   ⚠️  Training script not found")
else:
    print(f"   ℹ️  Repo not cloned yet (will clone when needed)")

# Check 6: Device availability
print("\n6️⃣  Checking device availability...")
import torch
if torch.cuda.is_available():
    gpu_name = torch.cuda.get_device_name(0)
    gpu_memory = torch.cuda.get_device_properties(0).total_memory / (1024**3)
    print(f"   ✅ GPU available: {gpu_name} ({gpu_memory:.1f} GB)")
else:
    print("   ℹ️  GPU not available (using CPU)")

print("\n" + "="*70)
print("VERIFICATION SUMMARY")
print("="*70)

# Final recommendations
recommendations = []

if base_ckpt.exists():
    checkpoint_dirs = [d for d in base_ckpt.iterdir() 
                      if d.is_dir() and d.name.startswith("checkpoints_")]
    if checkpoint_dirs:
        print(f"✅ You have {len(checkpoint_dirs)} condition(s) with checkpoints on Drive")
        print("✅ These checkpoints will persist across CPU/GPU sessions")
        print("✅ You can switch devices by changing --device cuda/cpu")
        print("\n📝 To resume training on CPU:")
        print("   --checkpoint-dir /content/drive/MyDrive/MaxSight/checkpoints_<condition>")
        print("   --resume")
        print("   --device cpu")
    else:
        recommendations.append("No checkpoints found - start training to create them")
else:
    recommendations.append("Base checkpoint directory not found - will be created on first training")

if recommendations:
    print("\n⚠️  Recommendations:")
    for rec in recommendations:
        print(f"   • {rec}")

print("\n" + "="*70)
print("✅ Setup check complete!")
print("="*70)
